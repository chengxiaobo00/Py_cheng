```js /templater 

let wv = app.workspace.getLeavesOfType("webviewer").sort(a=>-a.activeTime)[0]; 

let md = app.workspace.getLeavesOfType("markdown").sort(a=>-a.activeTime)[0]; 

if(wv && md){ 	
       
       let url = `[${wv.view.title}](${wv.view.webview.src}) `; 	
       
       await md.view.editor.replaceSelection(url); 
}
```
